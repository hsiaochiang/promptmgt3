```tsx
function Popover(props: {
  open: boolean;
  onClose: () => void;
  anchorRef: React.RefObject<HTMLElement>;
  children: React.ReactNode;
  width?: number;
}) {
  const panelRef = useRef<HTMLDivElement | null>(null);

  useEffect(() => {
    function onDoc(e: MouseEvent) {
      if (!props.open) return;
      const t = e.target as Node;
      if (panelRef.current?.contains(t)) return;
      if (props.anchorRef.current?.contains(t)) return;
      props.onClose();
    }
    document.addEventListener("mousedown", onDoc);
    return () => document.removeEventListener("mousedown", onDoc);
  }, [props.open, props.onClose, props.anchorRef]);

  if (!props.open) return null;

  const rect = props.anchorRef.current?.getBoundingClientRect();
  const style: React.CSSProperties = rect
    ? {
        position: "fixed",
        top: rect.bottom + 8,
        left: Math.min(rect.left, window.innerWidth - (props.width ?? 320) - 12),
        width: props.width ?? 320,
        zIndex: 50,
      }
    : { position: "fixed", top: 80, left: 80, width: props.width ?? 320, zIndex: 50 };

  return (
    <div ref={panelRef} style={style} className="rounded-xl border border-slate-200 bg-white shadow-sm p-2">
      {props.children}
    </div>
  );
}

function TagPicker(props: {
  allTags: string[];
  value: string[];
  onChange: (next: string[]) => void;
}) {
  const [q, setQ] = useState("");
  const ql = q.trim().toLowerCase();
  const options = useMemo(() => {
    const base = props.allTags.filter((t) => !ql || t.toLowerCase().includes(ql));
    // allow creating new tag
    if (ql && !props.allTags.some((t) => t.toLowerCase() === ql)) {
      return [`新增：「${q.trim()}」`, ...base];
    }
    return base;
  }, [props.allTags, ql, q]);

  return (
    <div className="p-2">
      <Input value={q} onChange={(e) => setQ(e.target.value)} placeholder="搜尋/新增標籤" />
      <div className="mt-2 max-h-56 overflow-auto rounded-lg border border-slate-200">
        {options.map((t) => {
          const isCreate = t.startsWith("新增：「");
          const label = isCreate ? q.trim() : t;
          const checked = props.value.includes(label);
          return (
            <button
              type="button"
              key={t}
              onClick={() => {
                if (!label) return;
                props.onChange(checked ? props.value.filter((x) => x !== label) : [...props.value, label]);
              }}
              className={cx(
                "w-full px-3 py-2 text-left text-sm",
                "border-b border-slate-200 last:border-b-0",
                "hover:bg-slate-50"
              )}
            >
              <div className="flex items-center justify-between">
                <span className={cx(isCreate ? "text-slate-900" : "text-slate-700")}>{isCreate ? t : label}</span>
                <span className="text-xs text-slate-400">{checked ? "已選" : ""}</span>
              </div>
            </button>
          );
        })}
        {!options.length ? <div className="px-3 py-3 text-sm text-slate-500">無結果</div> : null}
      </div>
      <div className="mt-2 text-xs text-slate-500">點選項目可切換；支援新增。</div>
    </div>
  );
}

function ClipboardPage(props: {
  snippets: Snippet[];
  onCreate: () => void;
  onUpdate: (id: string, patch: Partial<Snippet>) => void;
}) {
  const [activeId, setActiveId] = useState<string | null>(props.snippets[0]?.id ?? null);
  const active = props.snippets.find((s) => s.id === activeId) ?? null;

  const categories: Snippet["category"][] = ["角色設定", "回覆格式", "限制條件", "段落模板"];
  const [cat, setCat] = useState<Snippet["category"] | "全部">("全部");

  const list = useMemo(() => {
    const base = props.snippets.slice().sort((a, b) => b.updatedAt.localeCompare(a.updatedAt));
    return cat === "全部" ? base : base.filter((s) => s.category === cat);
  }, [props.snippets, cat]);

  useEffect(() => {
    if (list.length && !list.some((x) => x.id === activeId)) setActiveId(list[0].id);
  }, [cat]);

  return (
    <div className="h-full px-5 py-4">
      <div className="flex items-center justify-between">
        <div className="text-sm text-slate-600">剪貼簿：可儲存常用片語/段落模板，於提示詞編輯頁插入。</div>
        <BorderlessButton title="新增片語" onClick={() => props.onCreate()}>
          新建片語
        </BorderlessButton>
      </div>

      <div className="mt-3 grid gap-4" style={{ gridTemplateColumns: "260px 1fr" }}>
        <div className="rounded-2xl bg-white p-3 shadow-sm">
          <div className="mb-2 flex flex-wrap gap-1">
            <Pill label="全部" tone={cat === "全部" ? "blue" : "neutral"} onClick={() => setCat("全部")} />
            {categories.map((c) => (
              <Pill key={c} label={c} tone={cat === c ? "blue" : "neutral"} onClick={() => setCat(c)} />
            ))}
          </div>
          <div className="space-y-2">
            {list.map((s) => (
              <button
                key={s.id}
                type="button"
                onClick={() => setActiveId(s.id)}
                className={cx(
                  "w-full rounded-xl p-3 text-left border transition",
                  activeId === s.id ? "border-slate-300 bg-slate-50" : "border-transparent hover:border-slate-200 hover:bg-slate-50"
                )}
              >
                <div className="truncate text-sm font-semibold text-slate-900">{s.title}</div>
                <div className="mt-1 text-xs text-slate-500">
                  {s.category} · {s.updatedAt}
                </div>
              </button>
            ))}
            {!list.length ? <div className="rounded-xl bg-slate-50 p-3 text-xs text-slate-500">無資料</div> : null}
          </div>
        </div>

        <div className="rounded-2xl bg-white p-4 shadow-sm">
          {active ? (
            <>
              <div className="grid gap-3" style={{ gridTemplateColumns: "1fr 200px" }}>
                <Input
                  label="標題"
                  value={active.title}
                  onChange={(e) => props.onUpdate(active.id, { title: e.target.value })}
                  placeholder="例如：角色設定：解決方案顧問"
                />
                <div>
                  <div className="mb-1 text-xs text-slate-500">分類</div>
                  <select
                    className="w-full rounded-lg border border-slate-200 bg-white px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-slate-200"
                    value={active.category}
                    onChange={(e) => props.onUpdate(active.id, { category: e.target.value as Snippet["category"] })}
                  >
                    {categories.map((c) => (
                      <option key={c} value={c}>
                        {c}
                      </option>
                    ))}
                  </select>
                </div>
              </div>

              <div className="mt-3">
                <Textarea
                  label="內容"
                  value={active.content}
                  onChange={(e) => props.onUpdate(active.id, { content: e.target.value })}
                  rows={14}
                  placeholder="輸入可重用的片語或段落模板…"
                />
              </div>
            </>
          ) : (
            <div className="text-sm text-slate-600">請先選取左側片語。</div>
          )}
        </div>
      </div>
    </div>
  );
}

export default function App() {
  const [route, setRoute] = useState<Route>(() => parseHash());
  const [projects, setProjects] = useState<Project[]>(seedProjects);
  const [prompts, setPrompts] = useState<Prompt[]>(seedPrompts);
  const [snippets, setSnippets] = useState<Snippet[]>(seedSnippets);
  const [globalQuery, setGlobalQuery] = useState("");

  useEffect(() => {
    if (!window.location.hash) go("/list");
    const onHash = () => setRoute(parseHash());
    window.addEventListener("hashchange", onHash);
    return () => window.removeEventListener("hashchange", onHash);
  }, []);

  const projectById = useMemo(() => new Map(projects.map((p) => [p.id, p])), [projects]);
  const promptById = useMemo(() => new Map(prompts.map((p) => [p.id, p])), [prompts]);

  const allTags = useMemo(() => {
    const s = new Set<string>();
    prompts.forEach((p) => p.tags.forEach((t) => s.add(t)));
    projects.forEach((p) => p.tags.forEach((t) => s.add(t)));
    return Array.from(s).sort((a, b) => a.localeCompare(b));
  }, [prompts, projects]);

  const filteredPrompts = useMemo(() => {
    const q = globalQuery.trim().toLowerCase();
    if (!q) return prompts;
    return prompts.filter((p) => {
      const projectName = p.projectId ? projectById.get(p.projectId)?.title ?? "" : "";
      const hay = [p.title, p.summary ?? "", p.content, projectName, p.tags.join(" ")].join(" ").toLowerCase();
      return hay.includes(q);
    });
  }, [globalQuery, prompts, projectById]);

  const promptsByProject = useMemo(() => {
    const map = new Map<string, Prompt[]>();
    filteredPrompts.forEach((pr) => {
      const pid = pr.projectId ?? "__no_project__";
      if (!map.has(pid)) map.set(pid, []);
      map.get(pid)!.push(pr);
    });

    const ordered: Array<{ project: Project | null; prompts: Prompt[] }> = [];
    projects.forEach((p) => {
      const ps = (map.get(p.id) ?? []).slice().sort((a, b) => b.updatedAt.localeCompare(a.updatedAt));
      ordered.push({ project: p, prompts: ps });
    });

    if (map.get("__no_project__")?.length) {
      ordered.push({
        project: null,
        prompts: (map.get("__no_project__") ?? []).slice().sort((a, b) => b.updatedAt.localeCompare(a.updatedAt)),
      });
    }
    return ordered;
  }, [filteredPrompts, projects]);

  // Right detail selection (list/boards only)
  const [selectedPromptId, setSelectedPromptId] = useState<string | null>(null);
  const [selectedProjectId, setSelectedProjectId] = useState<string | null>(null);

  useEffect(() => {
    // reset selection on edit pages
    if (route.name === "project" || route.name === "prompt") {
      setSelectedPromptId(null);
      setSelectedProjectId(null);
    }
  }, [route.name]);

  // mutations
  function upsertPrompt(patch: Partial<Prompt> & { id: string }) {
    setPrompts((prev) => prev.map((p) => (p.id === patch.id ? { ...p, ...patch, updatedAt: patch.updatedAt ?? todayISO() } : p)));
  }
  function upsertProject(patch: Partial<Project> & { id: string }) {
    setProjects((prev) => prev.map((p) => (p.id === patch.id ? { ...p, ...patch } : p)));
  }
  function createPrompt(projectId?: string) {
    const id = `prm-${Math.random().toString(16).slice(2, 8)}`;
    const p: Prompt = {
      id,
      title: "新提示詞",
      projectId,
      status: "draft",
      priority: "medium",
      tags: [],
      updatedAt: todayISO(),
      content: "# 角色設定\n\n# 任務目標\n\n# 回應格式\n\n# 限制條件\n",
    };
    setPrompts((prev) => [p, ...prev]);
    go(`/prompts-edit/${id}`);
  }
  function createProject() {
    const id = `pjt-${Math.random().toString(16).slice(2, 8)}`;
    const p: Project = { id, title: "新專案", status: "planned", tags: [], dateRange: "", description: "" };
    setProjects((prev) => [p, ...prev]);
    go(`/projects/${id}`);
  }

  return (
    <div className="h-screen w-full bg-slate-50 text-slate-900">
      <AppShell
        route={route}
        globalQuery={globalQuery}
        setGlobalQuery={setGlobalQuery}
        onNav={(path) => {
          setSelectedPromptId(null);
          setSelectedProjectId(null);
          go(path);
        }}
      >
        {/* Main */}
        <div className="h-full">
          {route.name === "list" ? (
            <ProjectPromptListPage
              projects={projects}
              promptsByProject={promptsByProject}
              onOpenProject={(id) => {
                setSelectedProjectId(id);
                setSelectedPromptId(null);
              }}
              onOpenPrompt={(id) => {
                setSelectedPromptId(id);
                setSelectedProjectId(null);
              }}
              onGoProject={(id) => go(`/projects/${id}`)}
              onGoPrompt={(id) => go(`/prompts-edit/${id}`)}
              onCreatePrompt={createPrompt}
              onCreateProject={createProject}
            />
          ) : route.name === "prompts" ? (
            <PromptPage
              prompts={filteredPrompts}
              projectById={projectById}
              initialView={route.view ?? "list"} // sidebar click defaults to list
              onSelectPrompt={(id) => {
                setSelectedPromptId(id);
                setSelectedProjectId(null);
              }}
              onOpenPrompt={(id) => go(`/prompts-edit/${id}`)}
              onMoveStatus={(id, next) => upsertPrompt({ id, status: next, updatedAt: todayISO() })}
              onCreatePrompt={() => createPrompt(undefined)}
            />
          ) : route.name === "clipboard" ? (
            <ClipboardPage
              snippets={snippets}
              onCreate={() => {
                const id = `snp-${Math.random().toString(16).slice(2, 8)}`;
                const s: Snippet = { id, title: "新片語", category: "段落模板", updatedAt: todayISO(), content: "" };
                setSnippets((prev) => [s, ...prev]);
              }}
              onUpdate={(id, patch) =>
                setSnippets((prev) => prev.map((s) => (s.id === id ? { ...s, ...patch, updatedAt: todayISO() } : s)))
              }
            />
          ) : route.name === "project" ? (
            <ProjectEditPage
              project={projectById.get(route.projectId) ?? null}
              prompts={prompts.filter((p) => p.projectId === route.projectId)}
              onBack={() => go("/list")}
              onUpdate={(patch) => upsertProject({ id: route.projectId, ...patch })}
              onOpenPrompt={(id) => go(`/prompts-edit/${id}`)}
              onCreatePrompt={() => createPrompt(route.projectId)}
            />
          ) : route.name === "prompt" ? (
            <PromptEditPage
              prompt={promptById.get(route.promptId) ?? null}
              projects={projects}
              allTags={allTags}
              snippets={snippets}
              onBack={() => go("/list")}
              onUpdate={(patch) => upsertPrompt({ id: route.promptId, ...patch, updatedAt: todayISO() })}
              onInsertSnippet={(text) => {
                const p = promptById.get(route.promptId);
                if (!p) return;
                upsertPrompt({ id: p.id, content: `${p.content}\n\n${text}`.trim(), updatedAt: todayISO() });
              }}
            />
          ) : null}
        </div>

        {/* Right Detail Panel */}
        <DetailPanel
          route={route}
          selectedPrompt={selectedPromptId ? promptById.get(selectedPromptId) ?? null : null}
          selectedProject={selectedProjectId ? projectById.get(selectedProjectId) ?? null : null}
          projectById={projectById}
          allTags={allTags}
          onClose={() => {
            setSelectedPromptId(null);
            setSelectedProjectId(null);
          }}
          onGoPrompt={(id) => go(`/prompts-edit/${id}`)}
          onGoProject={(id) => go(`/projects/${id}`)}
          onUpdatePrompt={(patch) => selectedPromptId && upsertPrompt({ id: selectedPromptId, ...patch, updatedAt: todayISO() })}
          onUpdateProject={(patch) => selectedProjectId && upsertProject({ id: selectedProjectId, ...patch })}
        />
      </AppShell>
    </div>
  );
}
```

## A. 新增「操作提示 Banner」元件（一次性新增）

請在檔案中搜尋關鍵字：`function PageToolbar`  
在 `PageToolbar` **下方**緊接著貼上：

```tsx
function GuidanceBanner(props: {
  title: string;
  items: Array<{ label: string; detail?: string }>;
}) {
  return (
    <div className="mx-5 mt-4 rounded-xl border border-slate-200 bg-white px-4 py-3 shadow-sm">
      <div className="flex items-start gap-3">
        <div className="mt-0.5 grid h-8 w-8 place-items-center rounded-lg bg-slate-900 text-white">
          i
        </div>
        <div className="min-w-0 flex-1">
          <div className="text-sm font-semibold text-slate-900">{props.title}</div>
          <ul className="mt-2 space-y-1">
            {props.items.map((it, idx) => (
              <li key={idx} className="text-sm text-slate-700">
                <span className="font-medium">{it.label}</span>
                {it.detail ? <span className="text-slate-500">{` — ${it.detail}`}</span> : null}
              </li>
            ))}
          </ul>
        </div>
      </div>
    </div>
  );
}
```

## C. 下一步我建議的「產品級」精修順序（我可以繼續幫你做）
你這句「要更像產品級（更清楚 CTA 層級）」最關鍵的是一致性與可預期性。下一輪建議依序做：
